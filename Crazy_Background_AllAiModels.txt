/* ===== three-bg.js ===== */

document.addEventListener('DOMContentLoaded', () => {
    initThreeBackground();
});

function initThreeBackground() {
    if (typeof THREE === 'undefined') {
        console.warn('Three.js not loaded — skipping 3D background');
        return;
    }

    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

    const canvas = document.getElementById('bg-canvas');
    if (!canvas) return;

    // ============================================================
    // PASTE YOUR CHOSEN VARIATION CONFIG HERE
    // (replace the block below with any variation from Part 2)
    // ============================================================
    const CONFIG = {
        colors:      [0xd4a574],   // array of hex colors — shapes pick randomly
        shapeTypes:  ['torus', 'octahedron', 'icosahedron', 'tetrahedron'],
        shapeCount:  15,
        spread:      100,
        opacity:     0.15,
        cameraZ:     30,
        mouseFactor: 5,
        lerpFactor:  0.05,
        scrollDepth: 20,
        rotSpeed:    0.01,         // max rotation speed multiplier
        wireframe:   true
    };
    // ============================================================

    // --- Scene ---
    const scene  = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(
        75, window.innerWidth / window.innerHeight, 0.1, 1000
    );
    camera.position.setZ(CONFIG.cameraZ);

    const renderer = new THREE.WebGLRenderer({
        canvas, alpha: true, antialias: true
    });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.setSize(window.innerWidth, window.innerHeight);

    // --- Geometry library ---
    // Add/remove entries from this map freely.
    const GEO = {
        // ---- ORIGINAL FOUR ----
        torus:        () => new THREE.TorusGeometry(10, 3, 16, 100),
        octahedron:   () => new THREE.OctahedronGeometry(8),
        icosahedron:  () => new THREE.IcosahedronGeometry(7),
        tetrahedron:  () => new THREE.TetrahedronGeometry(8),

        // ---- NEW SIMPLE SHAPES ----
        cube:         () => new THREE.BoxGeometry(10, 10, 10),
            // Clean crisp cube — very satisfying in wireframe.
            // Looks like a floating crate or die.

        lowPolySphere:() => new THREE.SphereGeometry(8, 8, 6),
            // Low segment count = faceted, angular look.
            // Not a smooth ball — more like a geode or gem.

        cone:         () => new THREE.ConeGeometry(7, 14, 8),
            // Eight-sided cone. Tall and directional.
            // Adds variety because it's not symmetrical in all axes.

        cylinder:     () => new THREE.CylinderGeometry(5, 5, 14, 8),
            // Eight-sided cylinder / prism.
            // Simple and architectural-feeling.

        flatRing:     () => new THREE.RingGeometry(5, 9, 6),
            // Flat hexagonal ring lying in 3D space.
            // Looks like a portal or coin tumbling through air.

        dodecahedron: () => new THREE.DodecahedronGeometry(7),
            // 12-faced. Fuller/rounder than icosahedron.
            // Slightly organic compared to the others.

        torusKnot:    () => new THREE.TorusKnotGeometry(6, 1.5, 80, 10),
            // The pretzel / infinity knot shape.
            // Instantly recognisable, very dynamic-looking.
            // Use sparingly — one or two per scene max.

        pyramid:      () => new THREE.ConeGeometry(8, 10, 4),
            // Four-sided pyramid / triangle. Egyptian.
            // Completely different silhouette from the others.

        hexPrism:     () => new THREE.CylinderGeometry(7, 7, 10, 6),
            // Six-sided prism. Honeycomb-like.

        thinRing:     () => new THREE.TorusGeometry(9, 0.8, 8, 60),
            // A thin ring/halo — much more delicate than the fat torus.
            // Great as a counterpoint to bulkier shapes.
    };

// Build the shapes
    const shapes = [];
    const isMobile = window.innerWidth < 768 || (navigator.hardwareConcurrency || 4) < 4;
    const count    = isMobile ? Math.floor(CONFIG.shapeCount / 2) : CONFIG.shapeCount;

    const MIN_DIST  = 18;
    const MAX_TRIES = 30;

    function findOpenPosition(existing, spread, minDist, maxTries) {
        for (let attempt = 0; attempt < maxTries; attempt++) {
            const candidate = new THREE.Vector3(
                THREE.MathUtils.randFloatSpread(spread),
                THREE.MathUtils.randFloatSpread(spread),
                THREE.MathUtils.randFloatSpread(spread)
            );
            const tooClose = existing.some(s => s.position.distanceTo(candidate) < minDist);
            if (!tooClose) return candidate;
        }
        return null;
    }

    for (let i = 0; i < count; i++) {
        const typeName = CONFIG.shapeTypes[Math.floor(Math.random() * CONFIG.shapeTypes.length)];
        const geometry = GEO[typeName]();
        const color    = CONFIG.colors[Math.floor(Math.random() * CONFIG.colors.length)];
        const material = new THREE.MeshStandardMaterial({
            color,
            wireframe:   CONFIG.wireframe,
            transparent: true,
            opacity:     CONFIG.opacity
        });
        const mesh = new THREE.Mesh(geometry, material);

        const pos = findOpenPosition(shapes, CONFIG.spread, MIN_DIST, MAX_TRIES);
        if (!pos) continue;
        mesh.position.copy(pos);

        mesh.rotation.x = Math.random() * Math.PI;
        mesh.rotation.y = Math.random() * Math.PI;
        mesh.userData.rotSpeed = {
            x: (Math.random() - 0.5) * CONFIG.rotSpeed,
            y: (Math.random() - 0.5) * CONFIG.rotSpeed
        };
        mesh.userData.baseOpacity = CONFIG.opacity;
        scene.add(mesh);
        shapes.push(mesh);
    }

    // Lighting
    const keyLight  = new THREE.PointLight(CONFIG.colors[0], 1.2);
    keyLight.position.set(20, 20, 20);
    scene.add(keyLight);
    const fillLight = new THREE.PointLight(0xffffff, 0.4);
    fillLight.position.set(-20, -10, 10);
    scene.add(fillLight);
    const ambLight  = new THREE.AmbientLight(0xffffff, 0.3);
    scene.add(ambLight);

    // Mouse tracking
    let mouseX = 0, mouseY = 0, targetX = 0, targetY = 0;
    document.addEventListener('mousemove', (e) => {
        mouseX = (e.clientX / window.innerWidth)  *  2 - 1;
        mouseY = (e.clientY / window.innerHeight) * -2 + 1;
    });

    // Scroll depth
    window.addEventListener('scroll', () => {
        const t = window.pageYOffset /
            Math.max(1, document.documentElement.scrollHeight - window.innerHeight);
        camera.position.z = CONFIG.cameraZ + t * CONFIG.scrollDepth;
    });

    // Animation loop
    function animate() {
        requestAnimationFrame(animate);

        targetX += (mouseX - targetX) * CONFIG.lerpFactor;
        targetY += (mouseY - targetY) * CONFIG.lerpFactor;

        camera.position.x = targetX * CONFIG.mouseFactor;
        camera.position.y = targetY * CONFIG.mouseFactor;
        camera.lookAt(scene.position);

        shapes.forEach(s => {
            s.rotation.x += s.userData.rotSpeed.x;
            s.rotation.y += s.userData.rotSpeed.y;
        });

        renderer.render(scene, camera);
    }

    animate();

    // Resize
    window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });

    // Section pulse
    const obs = new IntersectionObserver((entries) => {
        entries.forEach(e => { if (e.isIntersecting) pulseShapes(); });
    }, { threshold: 0.2 });

    document.querySelectorAll('.section, section').forEach(s => obs.observe(s));

    function pulseShapes() {
        shapes.forEach((shape, i) => {
            setTimeout(() => {
                let t = 0;
                const base = shape.userData.baseOpacity;
                const id   = setInterval(() => {
                    t += 0.06;
                    shape.material.opacity = base + Math.sin(t * Math.PI) * (base * 0.6);
                    if (t >= 1) { clearInterval(id); shape.material.opacity = base; }
                }, 16);
            }, i * 50);
        });
    }

    // Public API for theme switching
    window.updateBgColors = function(hexArray) {
        shapes.forEach(s => {
            const c = hexArray[Math.floor(Math.random() * hexArray.length)];
            s.material.color.setHex(c);
        });
        if (keyLight) keyLight.color.setHex(hexArray[0]);
    };
}

PART 3 — ENGRAVED PATTERNS REFERENCE
================================================================
Quick library of CSS patterns for the #bg-pattern element.
Mix and match with any variation. All are intentionally very subtle.

Fine grid (most popular — clean and technical):
  background-image:
      linear-gradient(rgba(212,165,116,0.05) 1px, transparent 1px),
      linear-gradient(90deg, rgba(212,165,116,0.05) 1px, transparent 1px);
  background-size: 40px 40px;

Small dots (soft, organic):
  background-image:
      radial-gradient(circle, rgba(240,237,232,0.12) 1px, transparent 1px);
  background-size: 28px 28px;

Large dots (sparser, more refined):
  background-image:
      radial-gradient(circle, rgba(212,165,116,0.08) 1.5px, transparent 1.5px);
  background-size: 48px 48px;

Diagonal lines (directional energy):
  background-image:
      repeating-linear-gradient(
          45deg,
          rgba(212,165,116,0.04) 0px, rgba(212,165,116,0.04) 1px,
          transparent 1px, transparent 18px
      );

Crosshatch (strongest pattern — use with lowest opacity shapes):
  background-image:
      repeating-linear-gradient(
          45deg,
          rgba(212,165,116,0.03) 0px, rgba(212,165,116,0.03) 1px,
          transparent 1px, transparent 18px
      ),
      repeating-linear-gradient(
          -45deg,
          rgba(212,165,116,0.03) 0px, rgba(212,165,116,0.03) 1px,
          transparent 1px, transparent 18px
      );

Micro grid (very fine, almost invisible — like linen texture):
  background-image:
      linear-gradient(rgba(255,255,255,0.03) 1px, transparent 1px),
      linear-gradient(90deg, rgba(255,255,255,0.03) 1px, transparent 1px);
  background-size: 16px 16px;

Hex dots (triangular grid — more unusual):
  background-image:
      radial-gradient(circle, rgba(212,165,116,0.09) 1px, transparent 1px),
      radial-gradient(circle, rgba(212,165,116,0.09) 1px, transparent 1px);
  background-size: 24px 42px;
  background-position: 0 0, 12px 21px;

Scaling opacity:
  With any pattern, the RGB values and the alpha value are your two controls.
  Use the same RGB as your accent so it feels integrated, not foreign.
  Alpha 0.03 = barely there (felt, not seen)
  Alpha 0.06 = noticeable on close inspection
  Alpha 0.10 = clearly visible, part of the design
  Alpha 0.15+ = prominent, almost an intentional graphic element

================================================================
PART 4 — SHAPE NOTES
================================================================
Notes on how each shape reads in wireframe, to help you pick combinations.

torus          — The iconic donut ring. Familiar and satisfying. Good anchor shape.
octahedron     — Clean diamond. Symmetrical from every angle. Very geometric.
icosahedron    — Faceted "almost sphere". Sophisticated, not round, not sharp.
tetrahedron    — Simple triangle pyramid. The most minimal 3D shape.
cube           — Instantly readable. Good contrast against curved shapes. Feels solid.
lowPolySphere  — Rough faceted ball, like a gem or a mineral crystal. 8 segments wide.
cone           — Directional and asymmetric. Adds movement because it points.
cylinder       — Architectural, like a column or a drum. Very stable-looking.
flatRing       — 2D-feeling ring tumbling in 3D. Like a coin spinning. Very dynamic.
dodecahedron   — 12 pentagons. Rounder and more organic than icosahedron.
torusKnot      — The pretzel. Instantly striking. Use 1–2 max — it's a statement shape.
pyramid        — 4-sided cone. Egyptian/architectural. Very different silhouette.
hexPrism       — Six-sided column. Honeycomb/natural feeling.
thinRing       — Delicate halo. Much lighter presence than torus. Good "filler" shape.

Good combinations:
  Bold mix:       torus + cube + tetrahedron + cone           (sharp and varied)
  Organic mix:    icosahedron + dodecahedron + lowPolySphere  (round and mineral)
  Architectural:  cube + cylinder + pyramid + hexPrism        (structured and stable)
  Minimal:        torusKnot + thinRing + icosahedron          (complex vs delicate)
  Classic:        torus + octahedron + icosahedron + tetrahedron (the original 4)

================================================================
PART 6 — WHY IT WORKS (EXPLAIN TO ANOTHER AI)
================================================================

1. WIREFRAME + LOW OPACITY is the trick.
   Solid shapes at the same opacity = blobs. Wireframe = see-through lattice.
   You see the edges, the internal structure, and shapes behind each other.
   That layering creates perceived depth without any real depth sorting.

2. ALPHA: true ON THE RENDERER.
   The WebGL canvas background becomes transparent. The CSS body background
   shows through. Without this, you get a black WebGL rectangle.

3. MULTI-COLOR ARRAY.
   Randomly assigning each shape a color from the array makes the scene look
   designed and layered. Three colors (gold, white, dark grey) at the same
   opacity create the illusion of different depths because white reads as closer,
   dark grey reads as further away.

4. LERP FACTOR 0.03–0.05.
   The camera moves toward the mouse at (mouseX - currentX) * 0.05 per frame.
   This never fully reaches the target — so it always looks like it's catching up.
   That's the "floating on water" quality. Lower = dreamier. Higher = snappier.

5. SCROLL CAMERA PULL-BACK.
   Moving camera.position.z as the user scrolls makes shapes feel like they exist
   in a persistent world, not just a looping animation. The scene recedes as you
   explore the page. Subtle but deeply effective.

6. POINTER-EVENTS: NONE on the canvas.
   Without this, the canvas eats every click and the page is unusable.
   This is the most common mistake. Always set it.

7. THE ENGRAVED PATTERN LAYER.
   CSS gradient patterns are essentially free in terms of performance —
   they're static backgrounds rendered by the GPU once.
   The pattern layer sits between the body background and the canvas.
   It adds material texture to what would otherwise be a flat color void.
   The key is keeping the alpha low enough that it reads as "embedded" in
   the surface rather than drawn on top of it. 0.04–0.06 is usually right.

8. SHAPE COUNT VS SPREAD BALANCE.
   15 shapes at spread 100 = a medium-density field.
   6 shapes at spread 80 = sparse, each shape visible individually.
   25 shapes at spread 60 = dense, overlapping, more energetic.
   The density changes the entire personality of the background.
